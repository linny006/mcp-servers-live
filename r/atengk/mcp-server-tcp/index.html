Install & configure

Install method detected from README: uvx

uvx atengk-mcp-server-tcp

Claude Desktop config snippet

Add to your claude_desktop_config.json (~/Library/Application Support/Claude/claude_desktop_config.json on macOS, %APPDATA%\Claude\claude_desktop_config.json on Windows):

{
  "mcpServers": {
    "tcp": {
      "command": "uvx",
      "args": ["atengk-mcp-server-tcp"],
      "env": {
        "MCP_TCP_MAX_CONNECTIONS": "10",
        "MCP_TCP_IDLE_TIMEOUT_SECONDS": "300.0",
        "MCP_TCP_ALLOW_PRIVATE_NETWORKS": "true"
      }
    },
    "tcp-docker": {
      "command": "docker",
      "args": [
        "run",
        "-i",
        "--rm",
        "-e", "MCP_TCP_MAX_CONNECTIONS=10",
        "-e", "MCP_TCP_IDLE_TIMEOUT_SECONDS=300.0",
        "-e", "MCP_TCP_ALLOW_PRIVATE_NETWORKS=true",
        "ghcr.io/atengk/mcp-server-tcp:latest"
      ]
    }
  }
}
